# Telemetry Gateway (MQTT → decrypt → MongoDB)

### Goal

Subscribe to encrypted MQTT topics, decrypt the payload, and store it in MongoDB. You will need your solution from last lab. However, you can use an existing solution that is publishing as follows:

### 3) Create `gateway.py`

This version stores both:

- telemetry from `setu/<YOUR_TEAM_ID>/soil/telemetry`
- status from `setu/<YOUR_TEAM_ID>/soil/status`

```python
import json
import os
import socket
from datetime import datetime, timezone

import paho.mqtt.client as mqtt
from cryptography.fernet import Fernet
from pymongo import MongoClient

# -----------------------------
# Config
# -----------------------------
MQTT_HOST = os.getenv("MQTT_HOST", "test.mosquitto.org")
MQTT_PORT = int(os.getenv("MQTT_PORT", "8883"))
CA_CERT_PATH = os.getenv("CA_CERT_PATH", "./mosquitto.org.crt")

BASE_TOPIC = "setu/team1/soil"
TELEMETRY_TOPIC = BASE_TOPIC + "/telemetry"
STATUS_TOPIC = BASE_TOPIC + "/status"

MQTT_CLIENT_ID = os.getenv("MQTT_CLIENT_ID", f"setu_gateway_{socket.gethostname()}_{os.getpid()}")

FERNET_KEY = os.getenv("FERNET_KEY")
if not FERNET_KEY:
    raise RuntimeError("FERNET_KEY is not set")

cipher = Fernet(FERNET_KEY.encode("utf-8"))

MONGO_URL = os.getenv("MONGO_URL", "mongodb://localhost:27017")
MONGO_DB = os.getenv("MONGO_DB", "setu_soil")

mongo = MongoClient(MONGO_URL)
db = mongo[MONGO_DB]

telemetry_col = db["telemetry_events"]
status_col = db["status_events"]

def decrypt_payload(payload_bytes: bytes) -> dict:
    # payload is base64 token text (UTF-8)
    token_str = payload_bytes.decode("utf-8")
    plaintext = cipher.decrypt(token_str.encode("utf-8"))
    return json.loads(plaintext.decode("utf-8"))

# -----------------------------
# MQTT callbacks
# -----------------------------
def on_connect(client, userdata, flags, rc):
    if rc == 0:
        print(f"Gateway connected to MQTT (TLS)")
        client.subscribe([(TELEMETRY_TOPIC, 1), (STATUS_TOPIC, 1)])
        print("Subscribed to:", TELEMETRY_TOPIC, "and", STATUS_TOPIC)
    else:
        print("MQTT connect failed:", rc)

def on_disconnect(client, userdata, rc):
    if rc == 0:
        print("MQTT disconnected cleanly")
    else:
        print(f"MQTT disconnected unexpectedly (rc={rc})")

def on_message(client, userdata, msg):
        payload = decrypt_payload(msg.payload)

        if msg.topic == TELEMETRY_TOPIC:
            doc = {
                "device_id": payload.get("device_id", "unknown"),
                "soil_moisture": payload.get("soil_moisture"),
                "sensor_ts": payload.get("ts"),
                "telemetry_interval_seconds": payload.get("telemetry_interval_seconds"),
                "raw": payload,
            }
            telemetry_col.insert_one(doc)
            print("Stored telemetry:", doc["device_id"], doc["soil_moisture"])

def main():
    mqtt_client = mqtt.Client(client_id=MQTT_CLIENT_ID)
    mqtt_client.on_connect = on_connect
    mqtt_client.on_disconnect = on_disconnect
    mqtt_client.on_message = on_message

    mqtt_client.tls_set(CA_CERT_PATH)

    mqtt_client.connect(MQTT_HOST, MQTT_PORT, keepalive=60)
    mqtt_client.loop_forever()

if __name__ == "__main__":
    main()
```
