## Dashboard page (`static/index.html`)

Create the folder:

```bash
mkdir -p static
```

Create `static/index.html`:

```html
<!doctype html>
<html>
<head>
  <meta charset="utf-8" />
  <title>Soil Observability Dashboard</title>
  <script src="https://cdn.jsdelivr.net/npm/chart.js"></script>
  <style>
    body { font-family: system-ui, Arial; margin: 24px; }
    .row { display: flex; gap: 16px; flex-wrap: wrap; }
    .card { border: 1px solid #ddd; border-radius: 12px; padding: 16px; min-width: 240px; }
    .muted { color: #666; }
    .big { font-size: 28px; }
  </style>
</head>
<body>
  <h1>Soil Observability Dashboard</h1>

  <div class="row">
    <div class="card">
      <div class="muted">Current soil moisture</div>
      <div class="big" id="moisture">–</div>
      <div class="muted" id="moisture_ts">–</div>
    </div>

    <div class="card">
      <div class="muted">Relay state</div>
      <div class="big" id="relay">–</div>
      <div class="muted" id="relay_ts">–</div>
    </div>

    <div class="card">
      <div class="muted">Last refresh</div>
      <div id="refresh">–</div>
      <div class="muted">Polling every 5s</div>
    </div>
  </div>

  <h2>Soil moisture (last 30 minutes)</h2>
  <canvas id="chart" height="90"></canvas>

<script>
  const moistureEl = document.getElementById("moisture");
  const moistureTsEl = document.getElementById("moisture_ts");
  const relayEl = document.getElementById("relay");
  const relayTsEl = document.getElementById("relay_ts");
  const refreshEl = document.getElementById("refresh");

  function fmt(ts) {
    if (!ts) return "–";
    return new Date(ts * 1000).toLocaleString();
  }

  const ctx = document.getElementById("chart").getContext("2d");
  const chart = new Chart(ctx, {
    type: "line",
    data: { labels: [], datasets: [{ label: "Soil moisture", data: [] }] },
    options: { animation: false }
  });

  async function refreshLatest() {
    const r = await fetch("/api/latest");
    const data = await r.json();

    const t = data.telemetry;
    const s = data.status;

    if (t && t.soil_moisture !== undefined) {
      moistureEl.textContent = t.soil_moisture;
      moistureTsEl.textContent = "Updated: " + fmt(t.received_ts);
    } else {
      moistureEl.textContent = "–";
      moistureTsEl.textContent = "No telemetry yet";
    }

    if (s && s.state) {
      relayEl.textContent = s.state;
      relayTsEl.textContent = "Updated: " + fmt(s.received_ts);
    } else {
      relayEl.textContent = "–";
      relayTsEl.textContent = "No status yet";
    }
  }

  async function refreshChart() {
    const r = await fetch("/api/telemetry?minutes=30");
    const data = await r.json();

    const labels = data.rows.map(x => new Date(x.t * 1000).toLocaleTimeString());
    const values = data.rows.map(x => x.m);

    chart.data.labels = labels;
    chart.data.datasets[0].data = values;
    chart.update();
  }

  async function refreshAll() {
    try {
      await Promise.all([refreshLatest(), refreshChart()]);
      refreshEl.textContent = new Date().toLocaleTimeString();
    } catch (e) {
      refreshEl.textContent = "Error (check Flask console)";
      console.error(e);
    }
  }

  refreshAll();
  setInterval(refreshAll, 5000);
</script>

</body>
</html>
```
